SELENIUM - NOTES

LOCATORS
---------
id
name
className
tagName
linkText
partialLinkText
cssSelector
xpath

Example:
By.id("username")
By.name("login")
By.xpath("//*[@id='APjFqb']")

-------------------------

DROPDOWN
--------
Select s = new Select(element);

s.selectByVisibleText("India");
s.selectByIndex(2);
s.selectByValue("IN");

-------------------------

ACTION CLASS
------------
Actions action = new Actions(driver);

Drag & Drop
action.dragAndDrop(source,target).perform();

Drag by Offset
action.dragAndDropBy(element,x,y).perform();

Slider
action.dragAndDropBy(slider,50,0).perform();

Resize
action.dragAndDropBy(resizeHandle,x,y).perform();

-------------------------

WINDOW
------
driver.manage().window().maximize();

-------------------------

SWITCH TO
---------
driver.switchTo().alert();

driver.switchTo().frame("frameName");

driver.switchTo().window(windowID);

-------------------------

ALERT
-----
alert.accept();

alert.dismiss();

-------------------------

CLOSE VS QUIT
-------------
driver.close();   // current window

driver.quit();    // all browser windows
